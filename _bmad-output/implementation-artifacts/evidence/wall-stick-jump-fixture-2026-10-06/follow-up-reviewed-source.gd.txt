# --- Left and right wall runs (AC 1, 2; Tasks 1.2, 2.1, 2.3) -----------------


func test_wall_run_enters_once_on_both_sides_and_keeps_the_constraint_applied() -> void:
	var sixty_by_side := {}
	for tick_rate in [60, 120]:
		Engine.physics_ticks_per_second = tick_rate
		for side in [-1.0, 1.0]:
			var label := "side %s @ %d Hz" % [side, tick_rate]
			var report := await _wall_run_report(tick_rate, side)
			assert_true(report["ever_wall_running"], label)
			assert_eq(report["transition_counts"].get(LOCOMOTION_WALL_RUN, 0), 1, "entry must happen exactly once - %s" % label)
			assert_eq(report["active_steps"], report["constraint_applied_count"], "the wall constraint must resolve on every wall-run step - %s" % label)
			assert_gt(report["active_steps"], 0, label)
			assert_eq(report["commit_count_max"], 1, label)
			assert_almost_eq(report["wall_normal"].z, side, 0.05, label)
			assert_gt(report["wall_run_direction"].x, 0.9, label)
			assert_gt(report["end_along_wall_speed"], 0.0, "along-wall momentum vanished - %s" % label)
			if tick_rate == 60:
				sixty_by_side[side] = report
			else:
				# NFR4: same entry and along-wall speed at both rates; a wall
				# edge may be reached on a different diagnostic step.
				assert_eq(report["transition_counts"].get(LOCOMOTION_WALL_RUN, 0), sixty_by_side[side]["transition_counts"].get(LOCOMOTION_WALL_RUN, 0))
				assert_almost_eq(report["end_along_wall_speed"], sixty_by_side[side]["end_along_wall_speed"], 0.05)
			print("[1-9-wall-run] ", label, " -> ", report)
	Engine.physics_ticks_per_second = _saved_physics_ticks_per_second

func test_wall_run_direction_never_reverses_across_a_shallow_corner() -> void:

# --- Wall jump (AC 5; Tasks 5.1, 5.3, 5.4) -----------------------------------


func test_wall_jump_launches_once_and_leaves_an_ordinary_airborne_state() -> void:
	var at_sixty: Dictionary = {}
	for tick_rate in [60, 120]:
		Engine.physics_ticks_per_second = tick_rate
		var report := await _wall_jump_report(tick_rate)
		var label := "%d Hz" % tick_rate
		assert_true(report["ever_wall_running"], label)
		assert_eq(report["jump_count"], 1, "wall jump impulse applied more than once - %s" % label)
		assert_gt(report["jump_reference_outward_speed"], 0.0, "the jump must remain available with outward reference velocity - %s" % label)
		assert_almost_eq(report["launch_velocity"].y, 5.5, 0.02, label)
		assert_almost_eq(report["launch_velocity"].z, 8.0, 0.1, "authored away-from-wall launch changed - %s" % label)
		assert_almost_eq(
			report["launch_velocity"].x,
			report["pre_jump_along_wall_speed"],
			0.12,
			"the wall jump did not retain the incoming positive along-wall travel - %s" % label
		)
		# Task 5.3: the wall-jump exit leaves the wall run for ordinary airborne
		# movement on the very next step.
		assert_eq(report["state_at_jump"], LOCOMOTION_WALL_RUN, label)
		assert_eq(report["first_state_after_jump"], LOCOMOTION_AIRBORNE, label)
		assert_eq(report["commit_count_max"], 1, label)
		# AC 5: steering and grapple acquisition are available right away.
		assert_true(report["airborne_steering_available"], label)
		assert_true(report["grapple_acquisition_available"], label)
		assert_true(report["grapple_started_after_jump"], "grapple press after wall jump was ignored - %s" % label)
		if tick_rate == 60:
			at_sixty = report
		else:
			# One step of 4 m/s² acceleration may differ by 0.033 m/s.
			assert_lte((report["launch_velocity"] - at_sixty["launch_velocity"]).length(), 0.10)
			assert_eq(report["first_state_after_jump"], at_sixty["first_state_after_jump"])
		print("[1-9-wall-jump] ", label, " -> ", report)
	Engine.physics_ticks_per_second = _saved_physics_ticks_per_second

func test_low_speed_grapple_assisted_stick_keeps_zero_baseline_hold_and_release_at_both_rates() -> void:
	var configured_player: CharacterBody3D = PLAYER_SCENE.instantiate()
	var minimum := float(configured_player.get("wall_run_min_horizontal_speed"))
	var maximum := float(configured_player.get("wall_run_max_entry_speed"))
	configured_player.free()
	for tick_rate in [60, 120]:
		Engine.physics_ticks_per_second = tick_rate
		for initial_speed in [0.0, minimum * 0.5]:
			var label := "initial %s m/s @ %d Hz" % [initial_speed, tick_rate]
			var report := await _wall_stick_report(WallStickExit.RELEASE, tick_rate, initial_speed)
			var entry_velocity: Vector3 = report["entry_velocity"]
			assert_true(report["ever_wall_sticking"], label)
			assert_lt(Vector3(entry_velocity.x, 0.0, entry_velocity.z).length(), minimum, "entry was not genuinely subminimum: %s" % label)
			assert_gt(entry_velocity.length(), 0.0, "the real grapple pull must be present: %s" % label)
			assert_lte(entry_velocity.length(), maximum, label)
			assert_false(report["entry_run_gate_open"], label)
			assert_true(report["entry_speed_gate_open"], label)
			assert_eq(report["entry_speed_reason"], &"pass", label)
			assert_eq(report["transition_counts"].get(LOCOMOTION_WALL_STICK, 0), 1, label)
			assert_eq(report["transition_counts"].get(LOCOMOTION_WALL_RUN, 0), 0, label)
			assert_almost_eq((report["first_hold_initial_velocity"] as Vector3).length(), 0.0, 0.00001, "zero next-frame baseline: %s" % label)
			assert_eq(report["held_steps"], 18, label)
			assert_eq(report["held_steps"], report["hold_constraint_applied_count"], label)
			assert_almost_eq(report["maximum_hold_speed"], 0.0, 0.00001, label)
			assert_lte(report["maximum_hold_position_drift"], 0.001, label)
			assert_almost_eq(report["maximum_latch_position_drift"], 0.0, 0.00001, label)
			assert_true(report["exited"], label)
			assert_eq(report["transition_counts"].get(LOCOMOTION_AIRBORNE, 0), 1, label)
			assert_eq(report["post_exit_state"], LOCOMOTION_AIRBORNE, label)
			assert_false(report["still_wall_sticking"], label)
			assert_false(report["hold_request_on_exit"], label)
			assert_eq(report["hold_request_after_exit_count"], 0, label)
			assert_eq(report["terminal_reason"], GrappleEndReason.Reason.RELEASE, label)
			assert_eq(report["terminal_count"], 1, label)
			assert_true(report["wall_cleared_before_terminal_signal"], label)
			assert_true(report["duplicate_terminal_same"], label)
			assert_almost_eq((report["exit_velocity"] - report["pre_exit_velocity"]).length(), 0.0, 0.001, label)
			assert_eq(report["commit_count_max"], 1, label)
			print("[wall-stick-upper-low-speed] ", label, " -> entry=", entry_velocity, " horizontal=", Vector3(entry_velocity.x, 0.0, entry_velocity.z).length(), " total=", entry_velocity.length(), " baseline=", report["first_hold_initial_velocity"], " holds=", report["held_steps"], " stick_entries=", report["transition_counts"].get(LOCOMOTION_WALL_STICK, 0), " release_entries=", report["transition_counts"].get(LOCOMOTION_AIRBORNE, 0), " commits=", report["commit_count_max"])
	Engine.physics_ticks_per_second = _saved_physics_ticks_per_second

func test_wall_stick_rejects_outward_entry_and_cancels_an_outward_hold_once() -> void:
	var entry_report := await _wall_stick_outward_entry_report()
	assert_true(entry_report["valid_outward_entry_seen"], "the fixture never met the other wall-stick entry gates while separating")
	assert_false(entry_report["ever_wall_sticking"], "outward motion entered wall sticking")
	assert_eq(entry_report["transition_counts"].get(LOCOMOTION_WALL_STICK, 0), 0)
	assert_eq(entry_report["commit_count_max"], 1)

	var hold_report := await _wall_stick_report(WallStickExit.OUTWARD_VELOCITY)
	assert_true(hold_report["ever_wall_sticking"], "wall stick never engaged before the separation fixture")
	assert_true(hold_report["exited"], "outward velocity did not cancel the wall-stick hold")
	assert_eq(hold_report["transition_counts"].get(LOCOMOTION_AIRBORNE, 0), 1)
	assert_eq(hold_report["post_exit_state"], LOCOMOTION_AIRBORNE)
	assert_false(hold_report["still_wall_sticking"])
	assert_eq(hold_report["terminal_reason"], GrappleEndReason.Reason.STATE_CANCELLATION)
	assert_eq(hold_report["terminal_count"], 1)
	assert_true(hold_report["duplicate_terminal_same"])
	assert_true(hold_report["wall_cleared_before_terminal_signal"])
	assert_false(hold_report["hold_request_on_exit"])
	assert_eq(hold_report["hold_request_after_exit_count"], 0)
	assert_true(hold_report["valid_outward_hold_seen"], "the cancellation step did not have supported selected-wall contact")
	var outward_reference: Vector3 = hold_report["pre_exit_velocity"]
	var selected_wall_normal: Vector3 = hold_report["wall_normal"]
	var exit_velocity: Vector3 = hold_report["exit_velocity"]
	assert_almost_eq(
		(exit_velocity - outward_reference).length(),
		0.0,
		0.001,
		"outward cancellation must preserve the transaction's recoverable velocity"
	)
	assert_gt(outward_reference.dot(selected_wall_normal), 0.0, "the cancellation fixture must have positive outward speed")
	assert_eq(hold_report["commit_count_max"], 1)

	var release_report := await _wall_stick_report(WallStickExit.RELEASE_OUTWARD_VELOCITY)
	assert_true(release_report["exited"], "outward motion plus grapple release did not exit wall sticking")
	assert_eq(release_report["post_exit_state"], LOCOMOTION_AIRBORNE)
	assert_false(release_report["still_wall_sticking"])
	assert_false(release_report["grapple_held_at_exit"], "the cancellation case did not actually release Grapple")
	assert_true(release_report["valid_outward_hold_seen"], "the release case did not have supported selected-wall contact")
	assert_eq(release_report["terminal_reason"], GrappleEndReason.Reason.STATE_CANCELLATION)
	assert_eq(release_report["terminal_count"], 1)
	assert_false(release_report["hold_request_on_exit"])
	assert_eq(release_report["hold_request_after_exit_count"], 0)
	var release_pre_velocity: Vector3 = release_report["pre_exit_velocity"]
	var release_wall_normal: Vector3 = release_report["wall_normal"]
	var horizontal_release_normal := Vector3(
		release_wall_normal.x,
		0.0,

func _wall_run_report(tick_rate: int, side: float, entry_speed: float = 8.0) -> Dictionary:
	var fixture := _new_world(Vector3(-3.0, 2.0, 0.0))
	var player: CharacterBody3D = fixture["player"]
	var input_source: PlayerInputSource = fixture["input_source"]
	var motor: PlayerMotor = fixture["motor"]
	# Wall on the requested side: normal +Z for side = 1, normal -Z for side = -1.
	_add_box(fixture["world"], Vector3(0.0, 4.0, -side * 1.0), Vector3(40.0, 8.0, 0.2), 0.0)
	await _settle(motor)

	var delta := 1.0 / float(tick_rate)
	var steps := int(ceil(1.5 * float(tick_rate)))
	player.velocity = Vector3(entry_speed, 0.0, 0.0)
	input_source.inject_movement_strengths(0.0, 1.0, 0.0, 0.0)

	var report := _new_report()
	_begin(report, motor)
	for _i in range(steps):
		player.call("_physics_process", delta)
		_observe(report, player, motor)
		var result: PlayerMotorCommitResult = motor.get_last_commit_result()
		if result != null and result.locomotion_state_id == LOCOMOTION_AIRBORNE and bool(player.get("is_wall_running")):
			report["entry_along_wall_speed"] = result.submitted_velocity.dot(player.get("wall_run_direction"))
		if (
			result != null
			and result.locomotion_state_id == LOCOMOTION_WALL_RUN
			and bool(player.get("is_wall_running"))
		):
			# Steady wall-run movement step: base motion, gravity policy, and the
			# wall constraint all resolve through the semantic phases. The entry
			# step is the preserved airborne passthrough and the exit step is the
			# preserved `EVENT_WALL_RUN_FINISHED` passthrough - neither submits a
			# wall constraint, and neither is a wall-run movement step.
			report["active_steps"] = int(report["active_steps"]) + 1
			if _result_has_source(result, SOURCE_WALL_RUN_CONSTRAINT):
				report["constraint_applied_count"] = int(report["constraint_applied_count"]) + 1
			report["wall_normal"] = player.get("wall_normal")
			report["wall_run_direction"] = player.get("wall_run_direction")
			report["end_along_wall_speed"] = result.submitted_velocity.dot(player.get("wall_run_direction"))
			if report["first_wall_run_speed"] < 0.0:
				report["first_wall_run_speed"] = report["end_along_wall_speed"]
	return report


func _wall_run_outward_entry_report() -> Dictionary:
	var fixture := _new_world(Vector3(-3.0, 2.0, -0.25))
	var player: CharacterBody3D = fixture["player"]
	var input_source: PlayerInputSource = fixture["input_source"]
	var motor: PlayerMotor = fixture["motor"]
	_add_box(fixture["world"], Vector3(0.0, 4.0, -1.0), Vector3(40.0, 8.0, 0.2), 0.0)
	await _settle(motor)

	var report := _new_report()
	_begin(report, motor)
	input_source.inject_movement_strengths(0.0, 1.0, 0.0, 0.0)
	for _i in range(24):
		var previous_contact: ContactFrame = motor.get_previous_contact_frame()
		var outward_normal := Vector3(0.0, 0.0, 1.0)
		if previous_contact != null and previous_contact.has_wall_contact:
			outward_normal = previous_contact.wall_normal
		outward_normal.y = 0.0
		outward_normal = outward_normal.normalized()
		# Re-assert a tiny positive separating component each transaction so
		# ordinary airborne acceleration cannot turn this into an inward control.
		player.velocity = Vector3(8.0, 0.0, 0.0) + outward_normal * 0.01
		player.call("_physics_process", 1.0 / 60.0)
		_observe(report, player, motor)
		var result: PlayerMotorCommitResult = motor.get_last_commit_result()
		if result != null and result.contact_frame.has_wall_contact:
			var frame: ContactFrame = result.contact_frame
			var reference_velocity: Vector3 = player.call("get_motion_start_velocity")
			var command_frame: PlayerCommandFrame = player.call("get_player_command_frame")
			if command_frame == null:
				continue
			var run_direction: Vector3 = player.call(
				"_get_wall_run_direction",
				frame.wall_normal,
				reference_velocity
			)
			var other_entry_gates_open: bool = (
				not player.call("has_ground_contact")
				and not bool(player.get("is_grappling"))
				and bool(player.call("_is_runnable_wall_frame", frame, result.physics_step))
				and bool(player.call("_can_start_wall_run", reference_velocity))
				and bool(
					player.call(
						"_has_wall_run_input_for_direction",
						command_frame.movement_axis,
						run_direction
					)
				)
			)
			if (
				other_entry_gates_open
				and bool(player.call("_has_outward_wall_speed", reference_velocity, frame.wall_normal))
			):
				report["valid_outward_entry_seen"] = true
	return report


func _wall_run_outward_exit_report() -> Dictionary:
	var fixture := _new_world(Vector3(-3.0, 2.0, 0.0))
	var player: CharacterBody3D = fixture["player"]
	var input_source: PlayerInputSource = fixture["input_source"]
	var motor: PlayerMotor = fixture["motor"]
	_add_box(fixture["world"], Vector3(0.0, 4.0, -1.0), Vector3(40.0, 8.0, 0.2), 0.0)
	await _settle(motor)

	var report := _new_report()
	_begin(report, motor)
	player.velocity = Vector3(8.0, 0.0, 0.0)
	input_source.inject_movement_strengths(0.0, 1.0, 0.0, 0.0)
	for _i in range(30):
		player.call("_physics_process", 1.0 / 60.0)
		_observe(report, player, motor)
		var result: PlayerMotorCommitResult = motor.get_last_commit_result()
		if (
			result != null
			and result.locomotion_state_id == LOCOMOTION_WALL_RUN
			and bool(player.get("is_wall_running"))
		):
			break
	assert_true(bool(player.get("is_wall_running")), "the fixture never reached active wall running")

	var wall_normal: Vector3 = player.get("wall_normal")
	var horizontal_normal := Vector3(wall_normal.x, 0.0, wall_normal.z).normalized()
	var reference_velocity: Vector3 = player.call("get_committed_motion_velocity") + horizontal_normal * 0.5
	var contact_frame: ContactFrame = motor.get_previous_contact_frame()
	var command_frame: PlayerCommandFrame = player.call("get_player_command_frame")
	report["valid_outward_maintenance_seen"] = (
		bool(player.get("is_wall_running"))
		and not bool(player.call("has_ground_contact"))
		and not bool(player.get("is_grappling"))
		and contact_frame != null
		and command_frame != null
		and not command_frame.was_pressed(PlayerCommandFrame.Action.JUMP)
		and not command_frame.was_pressed(PlayerCommandFrame.Action.GRAPPLE)
		and bool(
			player.call(
				"_is_runnable_wall_frame",
				contact_frame,
				int(player.get("_player_physics_step"))
			)
		)
		and bool(
			player.call(
				"_has_wall_run_input_for_direction",
				command_frame.movement_axis,
				player.get("wall_run_direction")
			)
		)
		and bool(player.call("_has_outward_wall_speed", reference_velocity, wall_normal))
	)
	player.velocity = reference_velocity
	player.call("_physics_process", 1.0 / 60.0)
	_observe(report, player, motor)
	var exit_result: PlayerMotorCommitResult = motor.get_last_commit_result()
	report["exited"] = not bool(player.get("is_wall_running"))
	report["pre_exit_velocity"] = reference_velocity
	report["exit_velocity"] = exit_result.submitted_velocity
	report["wall_normal_after_exit"] = player.get("wall_normal")
	report["wall_constraint_on_exit"] = _result_has_source(exit_result, SOURCE_WALL_RUN_CONSTRAINT)

	player.call("_physics_process", 1.0 / 60.0)
	_observe(report, player, motor)
	var after_exit: PlayerMotorCommitResult = motor.get_last_commit_result()
	report["post_exit_state"] = after_exit.locomotion_state_id
	report["still_wall_running"] = bool(player.get("is_wall_running"))
	return report


func _corner_report(tick_rate: int) -> Dictionary:
	var fixture := _new_world(Vector3(-7.0, 2.0, 0.0))
	var player: CharacterBody3D = fixture["player"]
	var input_source: PlayerInputSource = fixture["input_source"]
	var motor: PlayerMotor = fixture["motor"]
	# Two overlapping faces with a 10-degree normal change: a shallow corner
	# that stays inside the authored 25-degree continuity tolerance.
	_add_box(fixture["world"], Vector3(-4.0, 4.0, -1.0), Vector3(12.0, 8.0, 0.2), 0.0)
	# Bring the oblique face inside the existing 0.8 m wall probe at the
	# overlap; the former -1.0 z center left an unsupported gap at 120 Hz.
	_add_box(fixture["world"], Vector3(3.0, 4.0, -0.65), Vector3(32.0, 8.0, 0.2), deg_to_rad(-10.0))
	await _settle(motor)

	var delta := 1.0 / float(tick_rate)
	# Cross the seam and follow the second face before the later sweep selects a
	# different wall relation (whose normal can correctly require a separation exit).
	var steps := int(ceil(1.45 * float(tick_rate)))
	player.velocity = Vector3(8.0, 0.0, 0.0)
	input_source.inject_movement_strengths(0.0, 1.0, 0.0, 0.0)

	var report := _new_report()
	_begin(report, motor)
	var first_direction := Vector3.ZERO
	var last_direction := Vector3.ZERO
	for _i in range(steps):
		var was_running: bool = bool(player.get("is_wall_running"))
		var velocity_before: Vector3 = player.velocity
		var prior_contact: ContactFrame = motor.get_previous_contact_frame()
		player.call("_physics_process", delta)
		_observe(report, player, motor)
		if was_running and not bool(player.get("is_wall_running")) and prior_contact != null:
			var horizontal_normal := Vector3(prior_contact.wall_normal.x, 0.0, prior_contact.wall_normal.z).normalized()
			var horizontal_velocity := Vector3(velocity_before.x, 0.0, velocity_before.z)
			report["outward_exit_projection"] = horizontal_velocity.dot(horizontal_normal)
			report["outward_exit_action"] = prior_contact.continuity_action
		if bool(player.get("is_wall_running")):
			report["active_steps"] = int(report["active_steps"]) + 1
			var direction: Vector3 = player.get("wall_run_direction")
			if last_direction != direction:
				report["corner_trace"].append({"x": player.global_position.x, "z": player.global_position.z, "direction": direction, "wall": player.get("wall_surface_identity")})
				last_direction = direction
			if first_direction == Vector3.ZERO:
				first_direction = direction
			elif first_direction.dot(direction) < report["min_direction_dot"]:
				report["min_direction_dot"] = first_direction.dot(direction)
	report["still_wall_running"] = bool(player.get("is_wall_running"))
	return report

	player.call("_physics_process", delta)
	_observe(report, player, motor)
	report["post_exit_state"] = motor.get_last_commit_result().locomotion_state_id
	report["still_wall_running"] = bool(player.get("is_wall_running"))
	return report

func _wall_jump_report(tick_rate: int) -> Dictionary:
	var fixture := _new_world(Vector3(-3.0, 2.0, 0.0))
	var player: CharacterBody3D = fixture["player"]
	var input_source: PlayerInputSource = fixture["input_source"]
	var motor: PlayerMotor = fixture["motor"]
	_add_box(fixture["world"], Vector3(0.0, 4.0, -1.0), Vector3(40.0, 8.0, 0.2), 0.0)
	await _settle(motor)

	var delta := 1.0 / float(tick_rate)
	player.velocity = Vector3(8.0, 0.0, 0.0)
	input_source.inject_movement_strengths(0.0, 1.0, 0.0, 0.0)

	var report := _new_report()
	_begin(report, motor)
	for _i in range(int(ceil(0.5 * float(tick_rate)))):
		player.call("_physics_process", delta)
		_observe(report, player, motor)
		if bool(player.get("is_wall_running")):
			break
	assert_true(bool(player.get("is_wall_running")), "never entered wall run @ %d Hz" % tick_rate)
	report["pre_jump_along_wall_speed"] = maxf(
		0.0,
		(player.call("get_committed_motion_velocity") as Vector3).dot(player.get("wall_run_direction"))
	)
	var wall_normal: Vector3 = player.get("wall_normal")
	var horizontal_normal := Vector3(wall_normal.x, 0.0, wall_normal.z).normalized()
	var jump_reference: Vector3 = player.call("get_committed_motion_velocity") + horizontal_normal * 0.5
	player.velocity = jump_reference
	report["jump_reference_outward_speed"] = jump_reference.dot(horizontal_normal)

	# Occurrence-identified jump, then keep the SAME command held for many
	# steps: neither the press-edge latch nor a second submission may apply
	# the impulse again (Task 5.1 double-apply guard).
	input_source.inject_action_binding(PlayerCommandFrame.Action.JUMP, 0, true)
	for _i in range(4):
		player.call("_physics_process", delta)
		_observe(report, player, motor)
		var result: PlayerMotorCommitResult = motor.get_last_commit_result()
		if result != null and _result_has_source(result, SOURCE_JUMP_WALL):
			report["jump_count"] = int(report["jump_count"]) + 1
			report["launch_velocity"] = result.submitted_velocity
			report["state_at_jump"] = result.locomotion_state_id
			break
	assert_eq(report["jump_count"], 1, "the fixture did not launch from the wall run")

	# The next command frame must accept airborne steering and evaluate real
	# grapple targeting against the wall, while the same jump is still held.
	input_source.inject_movement_strengths(1.0, 0.0, 0.0, 0.0)
	player.call("_physics_process", delta)
	_observe(report, player, motor)
	var steering_result: PlayerMotorCommitResult = motor.get_last_commit_result()
	report["first_state_after_jump"] = steering_result.locomotion_state_id
	report["airborne_steering_available"] = steering_result.submitted_velocity.x < report["launch_velocity"].x - 0.001
	var targeting: GrappleTargetingResult = player.call("get_latest_grapple_targeting_result")
	report["grapple_acquisition_available"] = targeting != null and targeting.is_accepted()
	# The next press must be usable too; a held jump must not add another
	# impulse even when traversal changes to grappling or wall sticking.
	input_source.inject_action_binding(PlayerCommandFrame.Action.GRAPPLE, 0, true)
	for _i in range(int(ceil(0.35 * float(tick_rate)))):
		player.call("_physics_process", delta)
		_observe(report, player, motor)
		var result: PlayerMotorCommitResult = motor.get_last_commit_result()
		if _result_has_source(result, SOURCE_JUMP_WALL):
			report["jump_count"] = int(report["jump_count"]) + 1
		if bool(player.get("is_grappling")):
			report["grapple_started_after_jump"] = true
	input_source.inject_action_binding(PlayerCommandFrame.Action.GRAPPLE, 0, false)
	input_source.inject_action_binding(PlayerCommandFrame.Action.JUMP, 0, false)
	report["post_exit_state"] = motor.get_last_commit_result().locomotion_state_id
	report["still_wall_running"] = bool(player.get("is_wall_running"))
	return report

func _wall_stick_outward_entry_report() -> Dictionary:
	var fixture := _new_world(Vector3(-3.0, 2.0, -0.25))
	var player: CharacterBody3D = fixture["player"]
	var input_source: PlayerInputSource = fixture["input_source"]
	var motor: PlayerMotor = fixture["motor"]
	var world: Node3D = fixture["world"]
	_add_box(world, Vector3(0.0, 4.0, -1.0), Vector3(40.0, 8.0, 0.2), 0.0)
	await _settle(motor)

	var previous_contact: ContactFrame = motor.get_previous_contact_frame()
	var outward_normal := Vector3(0.0, 0.0, 1.0)
	if previous_contact != null and previous_contact.has_wall_contact:
		outward_normal = previous_contact.wall_normal
	outward_normal.y = 0.0
	outward_normal = outward_normal.normalized()
	# Give the seed an outward world hit position as well as an outward target;
	# the helper's default seed point is on the inward side of this wall.
	var anchor_position := Vector3(0.0, 2.0, 1.0) + outward_normal * 4.0
	var anchor := _add_anchor(world, anchor_position)

	var report := _new_report()
	_begin(report, motor)
	assert_eq(
		_seed_grapple(player, anchor, anchor_position),
		GrappleController.CommitStatus.SUCCESS
	)
	input_source.inject_action_binding(PlayerCommandFrame.Action.GRAPPLE, 0, true)
	input_source.inject_movement_strengths(0.0, 1.0, 1.0, 0.0)
	for _i in range(24):
		var contact_before_step: ContactFrame = motor.get_previous_contact_frame()
		if contact_before_step != null and contact_before_step.has_wall_contact:
			outward_normal = contact_before_step.wall_normal
			outward_normal.y = 0.0
			outward_normal = outward_normal.normalized()
		player.velocity = Vector3(6.0, 0.0, 0.0) + outward_normal * 0.05
		player.call("_physics_process", 1.0 / 60.0)
		_observe(report, player, motor)
		var result: PlayerMotorCommitResult = motor.get_last_commit_result()
		if result == null or not result.contact_frame.has_wall_contact:
			continue
		var frame := result.contact_frame
		var run_direction: Vector3 = player.call(
			"_get_wall_run_direction",
			frame.wall_normal,
			result.submitted_velocity
		)
		var command_frame: PlayerCommandFrame = player.call("get_player_command_frame")
		var all_other_entry_gates_open := (
			result.locomotion_state_id == LOCOMOTION_GRAPPLING
			and bool(player.get("is_grappling"))
			and command_frame != null
			and command_frame.is_held(PlayerCommandFrame.Action.GRAPPLE)
			and not frame.is_grounded
			and bool(player.call("_is_runnable_wall_frame", frame, result.physics_step))
			and bool(player.call("_can_start_wall_run", result.submitted_velocity))
			and bool(player.call("_has_wall_run_input_for_direction", Vector2(1.0, 0.0), run_direction))
		)
		if (
			all_other_entry_gates_open
			and bool(player.call("_has_outward_wall_speed", result.submitted_velocity, frame.wall_normal))
		):
			report["valid_outward_entry_seen"] = true
	return report


func _wall_stick_report(exit_mode: int, tick_rate: int = 60, entry_speed: float = 6.0) -> Dictionary:
	# A rate setter does not change the inherited physics callback's cached
	# delta. Cross both boundaries BEFORE spawning; retain automatic settling
	# and the synchronous entry/hold/jump transactions below.
	Engine.physics_ticks_per_second = tick_rate
	await get_tree().process_frame
	await get_tree().physics_frame
	var fixture := _new_world(Vector3(-3.0, 2.0, -0.25))
	var player: CharacterBody3D = fixture["player"]
	var input_source: PlayerInputSource = fixture["input_source"]
	var motor: PlayerMotor = fixture["motor"]
	var world: Node3D = fixture["world"]
	var controller: GrappleController = player.get("_grapple_controller")
	var terminal_events := [0, true, -1]
	controller.attachment_ended.connect(
		func(_attachment_id: StringName, _reason: GrappleEndReason.Reason) -> void:
			terminal_events[0] = int(terminal_events[0]) + 1
			terminal_events[1] = bool(terminal_events[1]) and not bool(player.get("is_wall_sticking"))
			if int(terminal_events[2]) == -1:
				terminal_events[2] = int(player.get("_player_physics_step"))
	)
	# Acquisition and physical wall are separate. Approved stick entry replaces
	# the former target, so lifecycle exits operate on the current attachment.
	var wall := _add_box(world, Vector3(0.0, 4.0, -1.0), Vector3(40.0, 8.0, 0.2), 0.0)
	# This fixture pulls into the wall (negative against the selected +Z normal),
	# keeping the pre-existing hold/release cases inside the new separation gate.
	var anchor := _add_anchor(world, Vector3(0.0, 2.0, -2.0))
	await _settle(motor)

	var delta := 1.0 / float(tick_rate)
	var report := _new_report()
	report.merge({
		"requested_rate": tick_rate,
		"requested_delta": delta,
		"entry_step": -1,
		"first_hold": {},
		"first_failed_hold": {},
		"hold_checks_count": 0,
		"aligned_hold_count": 0,
		"valid_hold_count": 0,
		"jump_ready": false,
		"jump_readiness": {},
		"readiness_step": -1,
		"prepared_jump_step": -1,
		"launch_step": -1,
		"terminal_step": -1,
	})
	report["first_hold_initial_velocity"] = Vector3.ZERO
	report["maximum_hold_speed"] = 0.0
	report["maximum_hold_position_drift"] = 0.0
	report["maximum_latch_position_drift"] = 0.0
	_begin(report, motor)
	assert_eq(_seed_grapple(player, anchor), GrappleController.CommitStatus.SUCCESS)
	input_source.inject_action_binding(PlayerCommandFrame.Action.GRAPPLE, 0, true)
	player.velocity = Vector3(entry_speed, 0.0, 0.0)
	# Literal forward may be held with backward: net tangent remains unchanged.
	input_source.inject_movement_strengths(0.0, 1.0, 1.0, 1.0)

	for _i in range(30):
		player.call("_physics_process", delta)
		_observe(report, player, motor)
		if bool(player.get("is_wall_sticking")):
			break
	assert_true(bool(player.get("is_wall_sticking")), "wall stick never engaged (exit mode %d)" % exit_mode)
	report["entry_step"] = int(player.get("_player_physics_step"))
	report["wall_normal"] = player.get("wall_stick_normal")
	var entry_hold_position: Vector3 = player.get("wall_stick_position")
	report["entry_velocity"] = motor.get_last_commit_result().submitted_velocity
	report["entry_run_gate_open"] = bool(player.call("_can_start_wall_run", report["entry_velocity"]))
	report["entry_speed_gate_open"] = bool(player.call("is_wall_stick_speed_gate_open"))
	report["entry_speed_reason"] = player.call("get_wall_stick_speed_gate_reason_id")

	for _i in range(18):
		player.call("_physics_process", delta)
		_observe(report, player, motor)
		var hold_result: PlayerMotorCommitResult = motor.get_last_commit_result()
		if hold_result != null and hold_result.is_hold_request:
			if _i == 0:
				report["first_hold_initial_velocity"] = motor.get_frame_initial_velocity()
			report["maximum_hold_speed"] = maxf(report["maximum_hold_speed"], maxf(hold_result.committed_velocity.length(), player.velocity.length()))
			report["maximum_hold_position_drift"] = maxf(report["maximum_hold_position_drift"], (hold_result.position_after - entry_hold_position).length())
			report["maximum_latch_position_drift"] = maxf(report["maximum_latch_position_drift"], ((player.get("wall_stick_position") as Vector3) - entry_hold_position).length())
			report["hold_request_seen"] = true
			report["held_steps"] = int(report["held_steps"]) + 1
			if _result_has_source(hold_result, SOURCE_WALL_STICK_HOLD):
				report["hold_constraint_applied_count"] = int(report["hold_constraint_applied_count"]) + 1
		var hold_evidence := _wall_stick_hold_evidence(player, motor, delta)
		report["hold_checks_count"] = int(report["hold_checks_count"]) + 1
		if _i == 0:
			report["first_hold"] = hold_evidence
		if hold_evidence["delta_aligned"]:
			report["aligned_hold_count"] = int(report["aligned_hold_count"]) + 1
		if hold_evidence["ready"]:
			report["valid_hold_count"] = int(report["valid_hold_count"]) + 1
		else:
			# Retain the first loss immediately; no later airborne calls may mask it.
			report["first_failed_hold"] = hold_evidence
			break
	assert_true(report["hold_request_seen"], "the wall-stick hold was never requested")

	match exit_mode:
		WallStickExit.RELEASE:
			input_source.inject_action_binding(PlayerCommandFrame.Action.GRAPPLE, 0, false)
		WallStickExit.WALL_LOST, WallStickExit.WALL_LOST_WITH_JUMP, WallStickExit.LOST_WALL_JUMP_WITH_GRAPPLE_RELEASE:
			# The re-anchored physical target goes away: no hold or stale jump.
			wall.free()
			if exit_mode != WallStickExit.WALL_LOST:
				report["lost_frame_seen_while_sticking"] = bool(player.get("is_wall_sticking"))
				input_source.inject_action_binding(PlayerCommandFrame.Action.JUMP, 0, true)
				if exit_mode == WallStickExit.LOST_WALL_JUMP_WITH_GRAPPLE_RELEASE:
					input_source.inject_action_binding(PlayerCommandFrame.Action.GRAPPLE, 0, false)
		WallStickExit.JUMP, WallStickExit.JUMP_WITH_GRAPPLE_RELEASE:
			if not _prepare_wall_stick_jump(report, player, motor, input_source, exit_mode, delta):
				return _finish_wall_stick_report(report, player, motor, controller, terminal_events)
		WallStickExit.OUTWARD_VELOCITY, WallStickExit.RELEASE_OUTWARD_VELOCITY:
			var separation_normal: Vector3 = player.get("wall_stick_normal")
			var horizontal_separation_normal := Vector3(separation_normal.x, 0.0, separation_normal.z).normalized()
			var outward_reference: Vector3 = player.get("wall_stick_run_direction") * 2.0 + horizontal_separation_normal * 0.5
			var contact_before_exit: ContactFrame = motor.get_previous_contact_frame()
			var current_step := int(player.get("_player_physics_step"))
			var selected_contact_supported: bool = (
				contact_before_exit != null
				and bool(player.call("_is_runnable_wall_frame", contact_before_exit, current_step))
				and contact_before_exit.continuity_action != ContactFrame.ContinuityAction.SWITCHED
				and (
					not bool(player.get("wall_stick_identity_persistent"))
					or (
						contact_before_exit.wall_identity_persistent
						and contact_before_exit.wall_surface_identity == player.get("wall_stick_surface_identity")
					)
				)
			)
			var command_before_exit: PlayerCommandFrame = player.call("get_player_command_frame")
			report["valid_outward_hold_seen"] = (
				bool(player.get("is_wall_sticking"))
				and selected_contact_supported
				and command_before_exit != null
				and command_before_exit.is_held(PlayerCommandFrame.Action.GRAPPLE)
				and not command_before_exit.was_pressed(PlayerCommandFrame.Action.JUMP)
				and bool(player.call("_has_outward_wall_speed", outward_reference, separation_normal))
			)
			player.velocity = outward_reference
			if exit_mode == WallStickExit.RELEASE_OUTWARD_VELOCITY:
				input_source.inject_action_binding(PlayerCommandFrame.Action.GRAPPLE, 0, false)
		WallStickExit.INVALIDATE_ATTACHMENT:
			# Sampling termination must address the ACTIVE target, not the
			# historical acquisition target (its destruction is now harmless).
			controller.get_attachment().get_target().free()
		WallStickExit.ROTATE_CAMERA:
			# A held player is stationary; looking behind must not remove the
			# unchanged required wall from the shared contact provider's probes.
			input_source.inject_mouse_motion(Vector2(-PI / input_source.mouse_sensitivity, 0.0))
		WallStickExit.SWITCH_WALL:
			# A perpendicular, supported face replaces the wall at the fixed hold
			# position. Switching walls must not retain the old hold or jump normal.
			var second_wall := _add_box(
				world,
				Vector3(player.global_position.x + 0.75, 4.0, player.global_position.z),
				Vector3(0.2, 8.0, 10.0),
				0.0
			)
			await get_tree().physics_frame
			assert_true(is_instance_valid(second_wall))
			wall.free()

	for i in range(12):
		# The velocity the next commit starts from: a non-jump exit must carry
		# it through unchanged (Task 7.4 recoverable velocity).
		var pre_exit_velocity: Vector3 = (
			player.velocity
			if (
				exit_mode == WallStickExit.OUTWARD_VELOCITY
				or exit_mode == WallStickExit.RELEASE_OUTWARD_VELOCITY
			)
			else player.call("get_committed_motion_velocity")
		)
		player.call("_physics_process", delta)
		_observe(report, player, motor)
		var result: PlayerMotorCommitResult = motor.get_last_commit_result()
		if result != null and result.contact_frame != null:
			if exit_mode == WallStickExit.ROTATE_CAMERA:
				report["wall_contact_after_turn"] = result.contact_frame.has_wall_contact
			if exit_mode == WallStickExit.SWITCH_WALL and result.contact_frame.continuity_action == ContactFrame.ContinuityAction.SWITCHED:
				report["switched_frame_seen"] = true
				report["selected_switched_normal"] = result.contact_frame.wall_normal
			if (
				(
					exit_mode == WallStickExit.WALL_LOST_WITH_JUMP
					or exit_mode == WallStickExit.LOST_WALL_JUMP_WITH_GRAPPLE_RELEASE
				)
				and result.contact_frame.wall_contact_lost
				and bool(player.get("is_wall_sticking"))
			):
				report["lost_frame_seen_while_sticking"] = true
				input_source.inject_action_binding(PlayerCommandFrame.Action.JUMP, 0, true)
				if exit_mode == WallStickExit.LOST_WALL_JUMP_WITH_GRAPPLE_RELEASE:
					input_source.inject_action_binding(PlayerCommandFrame.Action.GRAPPLE, 0, false)
		if _result_has_source(result, SOURCE_JUMP_WALL):
			report["jump_count"] = int(report["jump_count"]) + 1
			if int(report["launch_step"]) == -1:
				report["launch_step"] = result.physics_step
		if not bool(player.get("is_wall_sticking")):
			if (
				exit_mode == WallStickExit.RELEASE_OUTWARD_VELOCITY
				or exit_mode == WallStickExit.JUMP_WITH_GRAPPLE_RELEASE
				or exit_mode == WallStickExit.LOST_WALL_JUMP_WITH_GRAPPLE_RELEASE
			):
				var exit_command: PlayerCommandFrame = player.call("get_player_command_frame")
				report["grapple_held_at_exit"] = (
					exit_command != null
					and exit_command.is_held(PlayerCommandFrame.Action.GRAPPLE)
				)
			report["exited"] = true
			report["exit_latency_steps"] = i + 1
			report["hold_request_on_exit"] = (
				result != null
				and (result.is_hold_request or result.applied_constraints.has(SOURCE_WALL_STICK_HOLD))
			)
			if result != null:
				report["exit_velocity"] = result.submitted_velocity
				report["pre_exit_velocity"] = pre_exit_velocity
			break
	report["still_wall_sticking"] = bool(player.get("is_wall_sticking"))
	player.call("_physics_process", delta)
	_observe(report, player, motor)
	var after_exit: PlayerMotorCommitResult = motor.get_last_commit_result()
	if _result_has_source(after_exit, SOURCE_JUMP_WALL):
		report["jump_count"] = int(report["jump_count"]) + 1
	if exit_mode == WallStickExit.ROTATE_CAMERA and after_exit.contact_frame != null:
		report["wall_contact_after_turn"] = after_exit.contact_frame.has_wall_contact
	report["post_exit_state"] = after_exit.locomotion_state_id
	report["hold_request_after_exit_count"] = int(
		after_exit.is_hold_request or after_exit.applied_constraints.has(SOURCE_WALL_STICK_HOLD)
	)

	return _finish_wall_stick_report(report, player, motor, controller, terminal_events)


func _wall_stick_hold_evidence(player: CharacterBody3D, motor: PlayerMotor, delta: float) -> Dictionary:
	# This is POST-commit N. The public has_supported_wall_contact() expects
	# PRE-commit N-1, so use the existing frame gate with the just-committed N.
	# No support sampling, new queries, or gameplay mutation belongs here.
	var current_step := int(player.get("_player_physics_step"))
	var result: PlayerMotorCommitResult = motor.get_last_commit_result()
	var contact: ContactFrame = motor.get_previous_contact_frame()
	var binding: WallStickAttachment = player.get("_wall_stick_attachment")
	var command: PlayerCommandFrame = player.call("get_player_command_frame")
	var controller: GrappleController = player.get("_grapple_controller")
	var cached_delta := player.get_physics_process_delta_time()
	var active := bool(player.get("is_wall_sticking"))
	var successful_hold := (
		result != null
		and result.success
		and result.physics_step == current_step
		and result.locomotion_state_id == LOCOMOTION_WALL_STICK
		and result.is_hold_request
		and not result.hold_carry_blocked
		and result.commit_count == 1
		and _result_has_source(result, SOURCE_WALL_STICK_HOLD)
		and result.applied_constraints.has(SOURCE_WALL_STICK_HOLD)
	)
	var supported := (
		contact != null
		and bool(player.call("_is_runnable_wall_frame", contact, current_step))
		and not contact.is_grounded
		and contact.continuity_action != ContactFrame.ContinuityAction.SWITCHED
		and binding != null
		and binding.status == WallStickAttachment.Status.VALID
		and binding.matches_contact(contact)
	)
	var held_command := (
		command != null
		and command.physics_step == current_step
		and command.is_held(PlayerCommandFrame.Action.GRAPPLE)
		and command.move_forward_held
		and not command.was_pressed(PlayerCommandFrame.Action.JUMP)
	)
	var normal: Vector3 = player.get("wall_stick_normal")
	return {
		"ready": active and successful_hold and supported and held_command and controller.has_active_attachment() and Vector3(normal.x, 0.0, normal.z).length_squared() > 0.0,
		"current_step": current_step,
		"engine_frame": Engine.get_physics_frames(),
		"in_physics": Engine.is_in_physics_frame(),
		"configured_rate": Engine.physics_ticks_per_second,
		"requested_delta": delta,
		"cached_delta": cached_delta,
		"delta_aligned": is_equal_approx(delta, cached_delta),
		"active": active,
		"successful_hold": successful_hold,
		"supported": supported,
		"held_command": held_command,
		"binding_status": binding.status if binding != null else -1,
		"contact_step": contact.physics_step if contact != null else -1,
		"result_step": result.physics_step if result != null else -1,
		"result_success": result != null and result.success,
		"hold_requested": result != null and result.is_hold_request,
		"hold_carry_blocked": result != null and result.hold_carry_blocked,
		"hold_source": _result_has_source(result, SOURCE_WALL_STICK_HOLD),
		"submitted_velocity": result.submitted_velocity if result != null else Vector3.ZERO,
		"committed_velocity": result.committed_velocity if result != null else Vector3.ZERO,
		"body_velocity": player.velocity,
	}


func _prepare_wall_stick_jump(
	report: Dictionary,
	player: CharacterBody3D,
	motor: PlayerMotor,
	input_source: PlayerInputSource,
	exit_mode: int,
	delta: float
) -> bool:
	var readiness := _wall_stick_hold_evidence(player, motor, delta)
	report["jump_readiness"] = readiness
	report["readiness_step"] = readiness["current_step"]
	report["prepared_jump_step"] = -1
	report["jump_ready"] = (
		readiness["ready"]
		and int(report["valid_hold_count"]) == 18
		and report["first_failed_hold"].is_empty()
	)
	assert_true(report["jump_ready"], "wall-stick fixture lost maintained support before jump preparation: first_loss=%s readiness=%s" % [report["first_failed_hold"], readiness])
	if not report["jump_ready"]:
		return false
	# Guard before either velocity overwrite or input injection: cleared stick
	# vectors must never fabricate an ordinary airborne transaction as a launch.
	var jump_normal: Vector3 = player.get("wall_stick_normal")
	var horizontal_jump_normal := Vector3(jump_normal.x, 0.0, jump_normal.z).normalized()
	var jump_reference: Vector3 = player.get("wall_stick_run_direction") * 3.0 + horizontal_jump_normal * 0.5
	report["prepared_jump_step"] = int(readiness["current_step"]) + 1
	player.velocity = jump_reference
	report["jump_reference_outward_speed"] = jump_reference.dot(horizontal_jump_normal)
	input_source.inject_action_binding(PlayerCommandFrame.Action.JUMP, 0, true)
	if exit_mode == WallStickExit.JUMP_WITH_GRAPPLE_RELEASE:
		input_source.inject_action_binding(PlayerCommandFrame.Action.GRAPPLE, 0, false)
	return true


func _finish_wall_stick_report(
	report: Dictionary,
	player: CharacterBody3D,
	motor: PlayerMotor,
	controller: GrappleController,
	terminal_events: Array
) -> Dictionary:
	report["still_wall_sticking"] = bool(player.get("is_wall_sticking"))
	var result: PlayerMotorCommitResult = motor.get_last_commit_result()
	if result != null:
		report["post_exit_state"] = result.locomotion_state_id
	var attachment: GrappleAttachment = player.call("get_grapple_attachment")
	if attachment != null and attachment.has_committed_terminal():
		var terminal := attachment.get_terminal()
		report["terminal_reason"] = terminal.reason
		report["duplicate_terminal_same"] = (
			controller.terminate(GrappleEndReason.Reason.STATE_CANCELLATION, int(player.call("get_motion_step"))) == terminal
		)
	report["terminal_count"] = int(terminal_events[0])
	report["wall_cleared_before_terminal_signal"] = bool(terminal_events[1])
	report["terminal_step"] = int(terminal_events[2])
	return report


func _wall_stick_jump_readiness_negative_report() -> Dictionary:
	Engine.physics_ticks_per_second = 60
	await get_tree().process_frame
	await get_tree().physics_frame
	var fixture := _new_world(Vector3(-3.0, 2.0, -0.25))

	var input_source: PlayerInputSource = fixture["input_source"]

		player.call("_physics_process", 1.0 / 60.0)
		if bool(player.get("is_wall_sticking")):
			break
	var ready_before_release := true
	for _i in range(18):
		player.call("_physics_process", 1.0 / 60.0)
		ready_before_release = ready_before_release and bool(_wall_stick_hold_evidence(player, motor, 1.0 / 60.0)["ready"])
	# A real release transaction clears the maintained support; no production
	# fields, result objects, motor guards, or geometry are overridden.
	input_source.inject_action_binding(PlayerCommandFrame.Action.GRAPPLE, 0, false)
	player.call("_physics_process", 1.0 / 60.0)
	var negative := {
		"ready_before_release": ready_before_release,
		"active_after_release": bool(player.get("is_wall_sticking")),
		"hold_after_release": motor.get_last_commit_result().is_hold_request,
		"checks": [],
	}
	# A separate real GutTest instance records the expected guard assertion in
	# its own summary (gut stays null), without failing the enclosing test.
	var probe: GutTest = load(get_script().resource_path).new()
	add_child_autofree(probe)
	for exit_mode in [WallStickExit.JUMP, WallStickExit.JUMP_WITH_GRAPPLE_RELEASE]:
		var report := {"valid_hold_count": 18, "first_failed_hold": {}}
		var velocity_before := player.velocity
		var step_before := int(player.get("_player_physics_step"))
		var command_before: PlayerCommandFrame = player.call("get_player_command_frame")
		var failures_before: int = probe.get_fail_count()
		var prepared: bool = probe.call("_prepare_wall_stick_jump", report, player, motor, input_source, exit_mode, 1.0 / 60.0)
		var check := {
			"mode": exit_mode,
			"prepared": prepared,
			"explicit_failures": probe.get_fail_count() - failures_before,
			"readiness": report["jump_readiness"],
			"prepared_jump_step": report["prepared_jump_step"],
			"velocity_before": velocity_before,
			"velocity_after": player.velocity,
			"step_before": step_before,
			"step_after": int(player.get("_player_physics_step")),
			"same_command": command_before == player.call("get_player_command_frame"),
		}
		player.call("_physics_process", 1.0 / 60.0)
		var command_after: PlayerCommandFrame = player.call("get_player_command_frame")
		check["jump_pressed_after_step"] = command_after.was_pressed(PlayerCommandFrame.Action.JUMP)
		check["jump_source_after_step"] = _result_has_source(motor.get_last_commit_result(), SOURCE_JUMP_WALL)
		negative["checks"].append(check)
	return negative


func _death_report(mode: String) -> Dictionary:
	var fixture := _new_world(Vector3(-3.0, 2.0, -0.25))
	var player: CharacterBody3D = fixture["player"]
	var input_source: PlayerInputSource = fixture["input_source"]
	var motor: PlayerMotor = fixture["motor"]
	var controller: GrappleController = player.get("_grapple_controller")
	var terminal_events := [0]
	controller.attachment_ended.connect(
		func(_attachment_id: StringName, _reason: GrappleEndReason.Reason) -> void:
			terminal_events[0] = int(terminal_events[0]) + 1
	)
	var wall := _add_box(fixture["world"], Vector3(0.0, 4.0, -1.0), Vector3(40.0, 8.0, 0.2), 0.0)
	await _settle(motor)

	var delta := 1.0 / 60.0
	var report := _new_report()
	_begin(report, motor)
	if mode == "wall_stick":
		assert_eq(_seed_grapple(player, wall), GrappleController.CommitStatus.SUCCESS)
		input_source.inject_action_binding(PlayerCommandFrame.Action.GRAPPLE, 0, true)
	player.velocity = Vector3(6.0, 0.0, 0.0)
	input_source.inject_movement_strengths(0.0, 1.0, 1.0 if mode == "wall_stick" else 0.0, 1.0 if mode == "wall_stick" else 0.0)

	for _i in range(30):
		player.call("_physics_process", delta)
		_observe(report, player, motor)
		if mode == "wall_run" and bool(player.get("is_wall_running")):
			break
		if mode == "wall_stick" and bool(player.get("is_wall_sticking")):
			break
	report["reached_mode"] = (
		bool(player.get("is_wall_running")) if mode == "wall_run" else bool(player.get("is_wall_sticking"))
	)

	player.call("_set_dead")
	player.call("_set_dead")
	for _i in range(4):
		player.call("_physics_process", delta)
		_observe(report, player, motor)
	report["terminal_count"] = int(terminal_events[0])
	var attachment: GrappleAttachment = player.call("get_grapple_attachment")
	if attachment != null and attachment.has_committed_terminal():
		report["terminal_reason"] = attachment.get_terminal().reason
	report["post_exit_state"] = motor.get_last_commit_result().locomotion_state_id
	report["still_wall_running"] = bool(player.get("is_wall_running"))
	report["still_wall_sticking"] = bool(player.get("is_wall_sticking"))
	return report


func _recovery_report(mistake: String) -> Dictionary:
	var fixture := _new_world(Vector3(-3.0, 2.0, 0.0))
	var player: CharacterBody3D = fixture["player"]
	var input_source: PlayerInputSource = fixture["input_source"]
	var motor: PlayerMotor = fixture["motor"]
	var wall := _add_box(fixture["world"], Vector3(0.0, 4.0, -1.0), Vector3(40.0, 8.0, 0.2), 0.0)
	await _settle(motor)

	var delta := 1.0 / 60.0
	var report := _new_report()
	_begin(report, motor)
	var scene_before: Node = get_tree().current_scene
	var player_instance_id := player.get_instance_id()
	match mistake:
		"missed_entry":
			# Enough along-wall speed, but input deliberately misses alignment.
			player.velocity = Vector3(8.0, 0.0, 0.0)
			input_source.inject_movement_strengths(0.0, 0.0, 1.0, 0.0)
		"early_exit":
			player.velocity = Vector3(8.0, 0.0, 0.0)
			input_source.inject_movement_strengths(0.0, 1.0, 0.0, 0.0)
		"short_release":
			assert_eq(_seed_grapple(player, wall), GrappleController.CommitStatus.SUCCESS)
			input_source.inject_action_binding(PlayerCommandFrame.Action.GRAPPLE, 0, true)
			player.velocity = Vector3(6.0, 0.0, 0.0)
			input_source.inject_movement_strengths(0.0, 1.0, 1.0, 1.0)
		"imperfect_wall_jump":
			player.velocity = Vector3(8.0, 0.0, 0.0)
			input_source.inject_movement_strengths(0.0, 1.0, 0.0, 0.0)

	var released_grapple := false
	for i in range(45):
		var pre_step_velocity: Vector3 = player.call("get_committed_motion_velocity")
		player.call("_physics_process", delta)
		_observe(report, player, motor)
		var committed: PlayerMotorCommitResult = motor.get_last_commit_result()
		if mistake == "imperfect_wall_jump" and _result_has_source(committed, SOURCE_JUMP_WALL):
			report["jump_count"] = int(report["jump_count"]) + 1
		match mistake:
			"missed_entry":
				if i == 4:
					report["entry_was_missed"] = not bool(player.get("is_wall_running"))
					# Correct the input after the miss and keep traversing.
					input_source.inject_movement_strengths(0.0, 1.0, 0.0, 0.0)
			"early_exit":
				if bool(player.get("is_wall_running")):
					# Let go of the wall early: stop aligning with the run.
					input_source.inject_movement_strengths(0.0, 0.0, 1.0, 0.0)
				elif report["ever_wall_running"] and not report["exited_wall_early"]:
					report["exited_wall_early"] = true
					report["pre_exit_velocity"] = pre_step_velocity
					report["exit_velocity"] = committed.submitted_velocity
			"short_release":
				if bool(player.get("is_wall_sticking")):
					input_source.inject_action_binding(PlayerCommandFrame.Action.GRAPPLE, 0, false)
				elif report["ever_wall_sticking"] and not bool(player.get("is_grappling")):
					released_grapple = true
			"imperfect_wall_jump":
				if bool(player.get("is_wall_running")):
					input_source.inject_action_binding(PlayerCommandFrame.Action.JUMP, 0, true)
	report["released_grapple"] = released_grapple
	# Recovery guarantee (AC 9): after the mistake a reachable surface, a valid
	# grapple target, or the remaining air-control option is still usable, and
	# nothing restarted the level.
	var last: PlayerMotorCommitResult = motor.get_last_commit_result()
	var reachable_surface := (
		last != null
		and last.contact_frame != null
		and (last.contact_frame.is_grounded or last.contact_frame.has_wall_contact)
	)
	var targeting: GrappleTargetingResult = player.call("get_latest_grapple_targeting_result")
	var grapple_target_available := targeting != null and targeting.is_accepted()
	# A reported surface alone is not a usable recovery option. Push against
	# the prior travel direction and require the committed movement to respond.
	# A wall-run input exit needs one step to enter airborne before air steering
	# resumes; ground and air respond within the same bounded window.
	input_source.inject_movement_strengths(1.0, 0.0, 0.0, 0.0)
	var recovery_action_worked := false
	var last_velocity: Vector3 = last.committed_velocity
	for _i in range(3):
		player.call("_physics_process", delta)
		_observe(report, player, motor)
		var steered: PlayerMotorCommitResult = motor.get_last_commit_result()
		if steered.submitted_velocity.x < last_velocity.x - 0.001:
			recovery_action_worked = true
		last_velocity = steered.committed_velocity
	report["recovery_action_worked"] = recovery_action_worked
	report["recovery_option_available"] = (
		bool(player.call("is_player_physics_active"))
		and recovery_action_worked
		and (reachable_surface or grapple_target_available or report["prev_state"] == LOCOMOTION_AIRBORNE)
	)
	report["level_restarted"] = (
		player.get_instance_id() != player_instance_id
		or get_tree().current_scene != scene_before
		or not player.is_inside_tree()
	)
	return report

func _begin(report: Dictionary, motor: PlayerMotor) -> void:
	var result: PlayerMotorCommitResult = motor.get_last_commit_result()
	if result != null:
		report["prev_state"] = result.locomotion_state_id


func _observe(report: Dictionary, player: CharacterBody3D, motor: PlayerMotor) -> void:
	if bool(player.get("is_wall_running")):
		report["ever_wall_running"] = true
	if bool(player.get("is_wall_sticking")):
		report["ever_wall_sticking"] = true
	var result: PlayerMotorCommitResult = motor.get_last_commit_result()
	if result == null:
		return
	if result.commit_count > int(report["commit_count_max"]):
		report["commit_count_max"] = result.commit_count
	var state: StringName = result.locomotion_state_id
	if state != report["prev_state"]:
		var counts: Dictionary = report["transition_counts"]
		counts[state] = int(counts.get(state, 0)) + 1
		report["transition_counts"] = counts
		report["prev_state"] = state


func _result_has_source(result: PlayerMotorCommitResult, source_id: StringName) -> bool:
	if result == null:
		return false
	for phase_entry in result.phase_intermediates:
		if not (phase_entry is Dictionary):
			continue
		for identity in phase_entry.get("accepted_sources", []):
			if identity is Dictionary and identity.get("source_id", &"") == source_id:
				return true
	return false


func _settle(motor: PlayerMotor) -> void:
	var found := false
	for _i in range(SETTLE_FRAMES):
		await get_tree().physics_frame
		var result: PlayerMotorCommitResult = motor.get_last_commit_result()
		if result != null and result.locomotion_state_id == LOCOMOTION_AIRBORNE:
			found = true
			break
	assert_true(found, "the fixture never reached the airborne movement state")


func _new_world(player_position: Vector3) -> Dictionary:
	var viewport := SubViewport.new()
	viewport.size = Vector2i(256, 256)
	viewport.world_3d = World3D.new()
	get_tree().root.add_child(viewport)
	autofree(viewport)

	var world := Node3D.new()
	viewport.add_child(world)
	_add_box(world, Vector3(0.0, -0.1, 0.0), Vector3(60.0, 0.2, 60.0), 0.0)

	var player: CharacterBody3D = PLAYER_SCENE.instantiate()
	player.set("capture_mouse_on_start", false)
	player.position = player_position
	world.add_child(player)

	var input_source: PlayerInputSource = player.get_node(^"PlayerInputSource")
	var motor: PlayerMotor = player.get_node(^"PlayerMotor")
	input_source.enable_test_input_seam()
	return {
		"viewport": viewport,
		"world": world,
		"player": player,
		"input_source": input_source,
		"motor": motor,
	}


func _add_box(
	world: Node,
	box_position: Vector3,
	size: Vector3,
	rotation_y: float = 0.0,
	rotation_x: float = 0.0
) -> StaticBody3D:
	var box_body := StaticBody3D.new()
	box_body.position = box_position
	box_body.rotation = Vector3(rotation_x, rotation_y, 0.0)
	world.add_child(box_body)
	var shape_node := CollisionShape3D.new()
	var shape := BoxShape3D.new()
	shape.size = size
	shape_node.shape = shape
	box_body.add_child(shape_node)
	return box_body


func _add_anchor(world: Node, anchor_position: Vector3) -> StaticBody3D:
	# A collision-less body: it is a valid grapple anchor (Story 1.6 empty-identity
	# static response) but it can never be selected as a supported wall.
	var anchor := StaticBody3D.new()
	anchor.position = anchor_position
	world.add_child(anchor)
	return anchor


func _seed_grapple(
	player: CharacterBody3D,
	target: Node3D,
	seed_hit_position: Vector3 = Vector3(0.0, 0.1, -1.0)
) -> GrappleController.CommitStatus:
	var controller: GrappleController = player.get("_grapple_controller")
	assert_not_null(controller)
	var seed := GrappleTargetSeed.new(
		&"",
		seed_hit_position,
		Vector3(0.0, 0.0, 1.0),
		GrappleTargetResponse.static_default(),
		weakref(target),
		Vector3.ZERO
	)
	var status := controller.commit_attachment(seed, int(player.call("get_motion_step")) + 1)
	if status == GrappleController.CommitStatus.SUCCESS:
		player.call("dispatch_locomotion_event", &"grapple_started")
	return status
