def now():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def digest(data):
    return hashlib.sha256(data).hexdigest()


def write(name, data):
    with (OUT / name).open("x", encoding="utf-8", newline="\n") as f:
        json.dump(data, f, indent=2)
        f.write("\n")


def assertion_blocks(text):
    lines = text.splitlines(keepends=True)
    blocks = []
    i = 0
    while i < len(lines):
        if re.match(r"\s*assert_\w+\(", lines[i]):
            block = lines[i]
            depth = lines[i].count("(") - lines[i].count(")")
            while depth > 0:
                i += 1
                block += lines[i]
                depth += lines[i].count("(") - lines[i].count(")")
            blocks.append(block)
        i += 1
    return blocks


action = sys.argv[1]

    checks = []
    def check(name, ok):

        check(f"matrix {i+1}: maintained holds and unchanged tolerances", all(report[key] == 18 for key in ["aligned_holds","valid_holds","held_steps","hold_checks","hold_constraints"]) and report["first_failed_hold"] == {} and report["maximum_hold_speed"] <= 0.00001 and report["maximum_hold_drift"] <= 0.001 and report["maximum_latch_drift"] <= 0.00001)
        check(f"matrix {i+1}: step tuple and authored launch", [report[key] for key in ["readiness_step","prepared_step","launch_step","terminal_step"]] == [21,22,22,22] and report["jump_count"] == 1 and report["reference"] > 0 and abs(report["tangent_reference"]) > 0 and close(report["exit_velocity"]["x"],0,0.001) and close(report["exit_velocity"]["y"],5.5,0.02) and close(report["exit_velocity"]["z"],8,0.1))
        check(f"matrix {i+1}: terminal and cleanup", report["terminal_count"] == 1 and report["terminal_reason"] == 4 and report["commit_count_max"] == 1 and report["duplicate_terminal_same"] and report["wall_cleared_before_terminal"] and report["post_exit_state"] == "player.locomotion.airborne" and not report["still_sticking"] and report["post_exit_holds"] == 0 and (report["mode"] != 9 or not report["grapple_held_at_exit"]))
    inactive, supported = result["inactive"], result["supported"]
    check("inactive actual-release fixture and both modes", inactive["ready_before_release"] and not inactive["active_after_release"] and not inactive["hold_after_release"] and [c["mode"] for c in inactive["checks"]] == [2,9])
    check("supported fixture and eight checks", supported["ready_before_cases"] and len(supported["checks"]) == 8 and supported["active_after_destruction"] and supported["step_after_destruction_checks"] == supported["destruction_step"])

        else:
            check(name + ": active, grapple remains held", ready["active"] and case["grapple_binding_held_after"])
            if case["case"] == "destroyed_support":
                check(name + ": same pre-transaction window, bound support invalid", not ready["supported"] and not ready["ready"] and case["step_before"] == supported["destruction_step"] == 21)
            elif case["case"] == "delta_mismatch":
                check(name + ": mismatched requested delta only", ready["supported"] and ready["result_success"] and not ready["ready"] and not ready["delta_aligned"] and not ready["commit_delta_aligned"] and close(ready["cached_delta"],1/60) and close(ready["commit_delta"],1/60) and close(ready["requested_delta"],1/120))
            else:

    check("supported final transaction has no jump or grapple release", not supported["jump_pressed_after_step"] and not supported["jump_source_after_step"] and supported["grapple_held_after_step"])
    check("eight original automatic/private-world fixture players", len(result["fixture_nodes"]) == 8 and all(n["automatic_physics_enabled"] and n["private_world"] for n in result["fixture_nodes"]))
    check("helper clean and all temporary objects freed", result["helper_assertion_failures"] == 0 and result["tracked_objects_freed"] == 10 and result["remaining_autofree"] == 0 and result["temporary_context_absent"])
    check("60Hz configured/cached runtime restored and editor stopped ready", result["restored_rate"] == 60 and close(result["restored_cached_delta"],1/60) and record["stop"]["stopped"] and record["state"]["readiness"] == "ready" and not record["state"]["is_playing"])
    logs = record["editor_logs"]

elif action == "index":
    runs = []
    for manifest in sorted(OUT.glob("follow-up-*-manifest.json")):
        for run in json.loads(manifest.read_text())["runs"]:
            clean = re.sub(r"\x1b\[[0-9;]*[A-Za-z]", "", (OUT / run["log"]).read_text(encoding="utf-8", errors="replace"))
            summary = clean[clean.rfind("= Run Summary"):]
            def count(name):
                m = re.search(r"^" + re.escape(name) + r"\s+(\d+|none)\s*$", summary, re.M)
                return 0 if m is None or m.group(1) == "none" else int(m.group(1))
            assertions = re.search(r"^Asserts\s+(\d+)(?:/(\d+))?\s*$", summary, re.M)
            totals = {"scripts":count("Scripts"), "tests":count("Tests"),
                      "passing_tests":count("Passing Tests"), "failing_tests":count("Failing Tests"),
                      "passing_assertions":int(assertions.group(1)),
                      "total_assertions":int(assertions.group(2) or assertions.group(1))} if assertions else {}
            failures = re.findall(r"^- (test_\w+)\s*\n(?=\s*\[Failed\])", summary, re.M)
            runs.append({"log":run["log"], "exit_code":run["exit_code"],
                         "source_sha256":run["source_sha256"], "totals":totals,
                         "failed_tests":failures,
                         "shutdown_objectdb_leak":"8 ObjectDB instances were leaked" in clean,
                         "shutdown_resource_leak":"1 resources still in use" in clean})
    intact = (OUT / "follow-up-barrier-intact.gd.txt").read_bytes()

    control = intact.replace(anchor, b"\tEngine.physics_ticks_per_second = tick_rate\n", 1)

elif action == "batch":
    for selection, repetitions in [("regression",5), ("target",10), ("guards",3),
                                   ("group",3), ("file",3), ("recursive",3)]:
        result = subprocess.run(["rtk", "proxy", "python", str(pathlib.Path(__file__).resolve()),

    args = ["--headless", "--path", str(ROOT), "-s", "res://addons/gut/gut_cmdln.gd"]
    if selection == "recursive":
        args += ["-gdir=res://tests/player", "-ginclude_subdirs"]
    else:
        args += ["-gtest=res://tests/player/locomotion/test_wall_traversal_integration.gd"]
        filters = {"regression":"test_wall_stick_jump_synchronizes_inherited_opposite_rate_callbacks",
                   "target":"test_wall_stick_jump_uses_only_authored_up_and_horizontal_away_motion",
                   "guards":"wall_stick_jump_readiness", "group":"wall_stick_"}
        if selection != "file":
            args += ["-gunit_test_name=" + filters[selection]]
    args += ["-gexit"]
    version = subprocess.run([EXE, "--version"], capture_output=True, text=True).stdout.strip()
    runs = []
    for i in range(1, repetitions + 1):
        log_name = f"{phase}-{selection}-{i:02d}.log"
        started = now()
        with (OUT / log_name).open("xb") as log:
            result = subprocess.run([EXE] + args, cwd=ROOT, stdout=log, stderr=subprocess.STDOUT)
        clean = re.sub(r"\x1b\[[0-9;]*[A-Za-z]", "", (OUT / log_name).read_text(encoding="utf-8", errors="replace"))
        summary = clean[clean.rfind("= Run Summary"):]
        run = {"started_at": started, "ended_at": now(), "executable": EXE, "version": version,

               "source_sha256": digest(SOURCE.read_bytes()), "summary": summary}
        runs.append(run)
        print(json.dumps({k: v for k, v in run.items() if k != "arguments"}, indent=2), flush=True)
    write(f"{phase}-{selection}-manifest.json", {"wrapper":"rtk proxy python", "runs": runs})
elif action in ("final", "check"):
    baseline = json.loads((OUT / "baseline-preservation.json").read_text())
    changed, missing = [], []
    for path, before in baseline["hashes"].items():
        file = ROOT / path
        if not file.is_file():
            missing.append(path)
        elif digest(file.read_bytes()) != before:
            changed.append(path)
    current = SOURCE.read_text()
    assertion_results = {}
    for snapshot in ["baseline-source.gd.txt", "follow-up-reviewed-source.gd.txt"]:
        blocks = assertion_blocks((OUT / snapshot).read_text())
        offset, absent = 0, []
        for block in blocks:
            found = current.find(block, offset)
            if found < 0:
                absent.append(block)
            else:
                offset = found + len(block)
        assertion_results[snapshot] = {"count":len(blocks), "missing_or_modified":absent}
    def frozen(data):
        return data.split(b"<frozen-after-approval", 1)[1].split(b"</frozen-after-approval>", 1)[0]
    check = subprocess.run(["rtk", "git", "diff", "--check"], capture_output=True, text=True, cwd=ROOT)
    head = subprocess.run(["rtk", "proxy", "git", "rev-parse", "HEAD"], capture_output=True, text=True, cwd=ROOT)
    before = (OUT / "follow-up-reviewed-source.gd.txt").read_text().splitlines(keepends=True)
    after = current.splitlines(keepends=True)
    delta = list(difflib.unified_diff(before, after, fromfile="reviewed-initial", tofile="follow-up-restored"))
    if action == "final":
        with (OUT / "follow-up-source-diff.patch.txt").open("x", encoding="utf-8", newline="\n") as f:
            f.writelines(delta)
    status = subprocess.run(["rtk", "proxy", "git", "status", "--short"], capture_output=True, text=True, cwd=ROOT)
    json_files = sorted(OUT.glob("*.json"))
    for file in json_files:
        json.loads(file.read_text(encoding="utf-8"))
    data = {"captured_at":now(), "protected_count":len(baseline["hashes"]),
            "unchanged_count":len(baseline["hashes"])-len(changed)-len(missing),
            "changed_paths":changed, "missing_paths":missing, "assertions":assertion_results,
            "only_permitted_source_changed":changed == [SOURCE.relative_to(ROOT).as_posix()] and not missing,
            "other_protected_files_unchanged":len(baseline["hashes"])-1 if changed == [SOURCE.relative_to(ROOT).as_posix()] and not missing else None,
            "frozen_block_unchanged":frozen(SPEC.read_bytes()) == frozen((OUT / "approved-spec.txt").read_bytes()),
            "spec_status":re.search(r"^status: '([^']+)'", SPEC.read_text(), re.M).group(1),
            "source_sha256":digest(SOURCE.read_bytes()), "diff_check_exit":check.returncode,
            "diff_check_stdout":check.stdout, "diff_check_stderr":check.stderr, "head":head.stdout.strip(),
            "source_delta_added":sum(s.startswith("+") and not s.startswith("+++") for s in delta),
            "source_delta_removed":sum(s.startswith("-") and not s.startswith("---") for s in delta),
            "json_files_parsed":len(json_files), "status":status.stdout}
    if action == "final":
        write("follow-up-final-preservation.json", data)
    print(json.dumps(data, indent=2))
else:
    raise SystemExit("Unknown action: " + action)
